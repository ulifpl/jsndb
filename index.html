Why JSNDB?

Eliminate strict schemas and heavy connection protocols.

🚀

No Servers

Runs inside your JVM with zero external dependencies. Pure serverless efficiency.

🧩

POJO Native

Save Java Objects directly. No mappers, no DTOs, no boilerplate code required.

⚡

Binary Indices

Properties mapped into an optimized B-Tree for blazing fast lookup without JSON parsing.

🔄

Circular Support

Handles complex cyclic relationships natively using lazy-loading proxies.

Persona.java 
public class Persona {
    @jsndbObjectId
    private Long id;
    private String name;
    private List<Animal> pets;
    // constructors, getters, setters...
}

// 1. Insert 100k Entities
for (int i = 0; i < COUNT; i++) {
    persona p = new persona("User " + i, i, new ArrayList<>());
    for (int k=0; k < 3; k++) {
        animal a = new animal("Pet " + k, "Breed", p);
        p.getMascota().add(a);
        js.persist(a);
    }
    js.persist(p);
}
js.commit();

// 2. Select by Property (Indexed)
qwery q = qwery.create(persona.class, "User 500", "name", comparators.equal);

SQLite Implementation
// 1. Complex Table Setup & Inserts
st.execute("CREATE TABLE persona (id INTEGER PRIMARY KEY, name TEXT, age INTEGER)");
st.execute("CREATE TABLE animal (id INTEGER PRIMARY KEY, nombre TEXT, owner_id INTEGER)");
conn.setAutoCommit(false);

for (int i = 0; i < COUNT; i++) {
    psPersona.setString(2, "User " + i);
    psPersona.executeUpdate();
    for (int k=0; k < 3; k++) {
        psAnimal.setString(1, "Pet " + k);
        psAnimal.setInt(2, i);
        psAnimal.executeUpdate();
    }
}
conn.commit();

// 2. SQL Querying
PreparedStatement sel = conn.prepareStatement("SELECT * FROM persona WHERE name = ?");
sel.setString(1, "User 500");
ResultSet rs = sel.executeQuery();
Nitrite Implementation
// 1. Repository Setup & Indexing
ObjectRepository<persona> repo = db.getRepository(persona.class);
repo.createIndex("name", IndexOptions.indexOptions(IndexType.NonUnique));

for (int i = 0; i < COUNT; i++) {
    persona p = new persona("User " + i, i, new ArrayList<>());
    // Nitrite crashes on circular references without extra config
    repo.insert(p);
}
db.commit();

// 2. Object Filtering
Cursor<persona> cursor = repo.find(ObjectFilters.eq("name", "User 500"));
